// 1. Install: npm install @labsadik/universal-player
import { createPlayer } from '@labsadik/universal-player';
import '@labsadik/universal-player/style.css';

// 2. Put <div id="video-player"></div> in your HTML
const player = createPlayer('#video-player', {
  source: 'https://interactive-examples.mdn.mozilla.net/media/cc0-videos/flower.mp4',
  title: 'Product demo',
  theme: 'system', // 'dark' | 'light' | 'system'
  glass: true,
  accentColor: 'rgb(54, 214, 176)',
  controlsStyle: 'standard',
});

// 3. Optional runtime controls
player.setTheme('dark');
player.setFeatures({ stats: false, miniPlayer: false });

// 4. Clean up when your view is removed
player.destroy();
⎇ main*UTF-8TypeScript Ready

Lifecycle rule: call player.destroy() when the component unmounts.
